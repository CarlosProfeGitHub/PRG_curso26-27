    /*
 * Programa: ResumenUnidad01
 * Descripción: variables, constantes, operadores, conversiones
 *              y entrada por teclado
 * Autor: Ana Pérez
 * Fecha: 2026-09-15
 */
import java.util.Scanner;

public class ResumenUnidad01 {

    // Constante de clase: no cambia nunca
    static final double IVA = 0.21;

    public static void main(String[] args) {

        // ---------- 1. Declaración de variables ----------
        String nombre;
        int edad;
        double altura = 1.75;
        boolean aprobado;

        // ---------- 2. Lectura por teclado ----------
        Scanner teclado = new Scanner(System.in);

        System.out.print("Introduce tu nombre: ");
        nombre = teclado.nextLine();

        System.out.print("Introduce tu edad: ");
        edad = teclado.nextInt();
        teclado.nextLine();      // limpieza del búfer

        // ---------- 3. Constantes y cálculos ----------
        final double PRECIO_BASE = 100.0;
        double precioFinal = PRECIO_BASE + PRECIO_BASE * IVA;

        // ---------- 4. Operadores ----------
        aprobado = edad >= 18 && altura > 1.60;

        // ---------- 5. Conversión de tipo ----------
        int edadAproximada = (int) altura;   // 1 (trunca, no redondea)

        // ---------- 6. Salida ----------
        System.out.println("---- FICHA ----");
        System.out.println("Nombre    : " + nombre);
        System.out.println("Edad      : " + edad);
        System.out.println("Altura    : " + altura + " m");
        System.out.printf("Precio final: %.2f euros (IVA %.0f%%)%n", precioFinal, IVA * 100);
        System.out.println("Altura como entero: " + edadAproximada);
        System.out.println("¿Aprobado? " + aprobado);

        // ---------- 7. Números aleatorios ----------
        int dado = (int) (Math.random() * 6) + 1;
        System.out.println("Tirada de dado: " + dado);

        teclado.close();
    }
}
